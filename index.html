class Transaction {
    private $db;
    
    public function __construct() {
        $this->db = new Database();
    }
    
    public function process($from, $to, $amount) {
        // Валидация данных
        $this->validate($from, $to, $amount);
        
        // Начало транзакции
        $this->db->beginTransaction();
        
        try {
            // Списание
            $this->withdraw($from, $amount);
            // Зачисление
            $this->deposit($to, $amount);
            // Логирование
            $this->logTransaction($from, $to, $amount);
            
            $this->db->commit();
            return ['success' => true];
        } catch (Exception $e) {
            $this->db->rollback();
            return ['error' => $e->getMessage()];
        }
    }
}
